### 浏览量计数：内存累计 \+ 30 秒批量写库（第 288\~363 行）

```
_VC_COUNTER: dict = {}                    # 内存计数器 {文章id: 待写入的增量}
_VC_LOCK = _threading.Lock()              # 线程锁：多线程同时 +1 会互相覆盖
_VC_STOP = _threading.Event()             # 退出信号：Event.set() 可唤醒等待线程
_VC_FLUSH_INTERVAL = 30                   # 每 30 秒批量落盘一次

def _vc_count_view(post_id):              # 详情页访问时调用
    if not isinstance(post_id, int) or post_id <= 0:  # 参数守卫
        return
    with _VC_LOCK:                        # 加锁防止并发写坏字典
        _VC_COUNTER[post_id] = _VC_COUNTER.get(post_id, 0) + 1   # 内存 +1

def _vc_flush_now():                      # 把累计增量写回数据库
    snapshot = {}
    with _VC_LOCK:
        if _VC_COUNTER:
            snapshot = dict(_VC_COUNTER)  # 取出快照
            _VC_COUNTER.clear()           # 清空计数器（下次重新累计）
    if not snapshot:
        return
    try:
        with app.app_context():           # 数据库操作必须包 app 上下文
            cases = [f'WHEN {pid} THEN view_count + {inc}' ...]  # 拼 CASE WHEN
            sql = f'UPDATE post SET view_count = CASE id {" ".join(cases)} ELSE view_count END WHERE id IN (...)'
            db.session.execute(db.text(sql))   # 一条 SQL 批量更新所有文章
            db.session.commit()
    except Exception as exc:
        ...
        for pid, inc in snapshot.items(): # 失败则把增量放回内存，避免丢计数
            _VC_COUNTER[pid] = _VC_COUNTER.get(pid, 0) + inc

def _vc_flusher_loop():                   # 后台线程主循环
    while not _VC_STOP.is_set():          # 没收到停止信号就一直跑
        _vc_flush_now()                   # 先 flush 一次
        _VC_STOP.wait(_VC_FLUSH_INTERVAL) # 再等 30 秒（被 set 会立即醒来）

```

**为什么这么设计（性能核心）：**SQLite 是「写锁串行」数据库——任何写操作都会锁住整个库，期间所有读请求排队。如果每次访问文章都立刻 `commit()` 一次浏览量，高并发下页面会卡死。改成「内存 \+1、30 秒批量一条 SQL 更新」，写库次数从「每访问一次写一次」降到「每 30 秒写一次」。**删掉后果：**流量一大页面立刻卡；同时这也保护了 SSD 寿命。**注意 ****`_VC_STOP.wait()`**** 的妙用：**退出时 `_VC_STOP.set()` 会立刻打断 30 秒等待，循环马上再 flush 一次，保证最后 30 秒的浏览量不丢。

扩展知识点：**CASE WHEN 批量更新**是「一条 SQL 更新多行不同值」的技巧，`UPDATE post SET view_count = CASE id WHEN 1 THEN view_count+1 WHEN 2 THEN view_count+3 ELSE view_count END WHERE id IN (1,2)`——比循环 100 次单条 UPDATE 快一个数量级。

```
@app.after_request                       # 每个响应返回前执行
def _static_cache_control(response):
    path = request.environ.get('PATH_INFO') or request.path or ''
    is_static = (path.startswith('/static/') or path.startswith('/avatars/')
                 or path.startswith('/bg/'))
    if is_static:
        response.headers['Cache-Control'] = 'public, max-age=604800, immutable'
        # 静态资源缓存 7 天 + immutable（浏览器不再请求服务器，直接本地取）

@app.teardown_appcontext                 # 每个请求结束后执行
def shutdown_session(exception=None):
    if exception is not None:            # 请求抛异常了
        db.session.rollback()            # 回滚数据库事务，防止脏数据
    db.session.remove()                  # 释放会话（连接还回连接池）

@app.errorhandler(Exception)             # 兜底：任何未捕获异常
def handle_all_exception(e):
    code = getattr(e, 'code', 500)       # 拿 HTTP 状态码，拿不到按 500
    if isinstance(code, int) and code >= 500:
        app.logger.error('HTTP %s 未捕获异常 | URL=%s ...\n%s',
                         code, request.path, request.method, ..., traceback.format_exc())
        # 记录完整堆栈，方便事后排查
    ...
    return render_template(f'errors/{code}.html', error=str(e)[:200], code=code), code
    # 返回对应错误码的友好页面；模板缺失则逐级降级

```

**逐行要点：**① `immutable` 是缓存最强指令——静态资源 7 天内浏览器完全不发请求，加速二次访问；动态页面不加缓存，否则 flash 提示会被浏览器缓存导致重复显示；② `teardown` 里 `rollback + remove` 是「数据库卫生」：异常时不回滚，下个请求会复用坏掉的会话，最终「连接耗尽」；③ 兜底异常处理器把**完整堆栈写进日志**而只把「前 200 字符的错误信息」给用户看——既方便排查又不泄露内部细节。

```
    _vc_count_view(post.id)                     # 浏览量内存 +1（30 秒后批量写库，零阻塞）

    if isinstance(post.rendered_html, str) and post.rendered_html.strip():
        html = post.rendered_html               # 命中缓存列：跳过渲染
    if not html:
        html = render_markdown(post.content or '')    # 缓存为空 → 现场渲染（几十 ms）
        try:

                {'rendered_html': html}, synchronize_session=False)   # 跳过会话同步更快
            db.session.commit()
        except Exception:
            db.session.rollback()               # 回填失败不影响本次响应

        Post.id &lt; post.id, Post.published == True
    ).order_by(Post.id.desc()).first()          # 取最大的那个（离当前最近）
    next_post = Post.query.filter(              # 下一篇：id 更大且已发布
        Post.id &gt; post.id, Post.published == True
    ).order_by(Post.id.asc()).first()           # 取最小的那个

    # ---- 评论批量查询（1 条 SQL 拿全部评论，Python 内存分组）----
    # 旧实现是 N+1 查询：先查顶级评论，再逐条查回复 → 评论多时卡顿
    all_visible = (Comment.query
                   .filter_by(post_id=post.id, is_visible=True)   # 一次拿全部可见评论
                   .order_by(Comment.created_at.asc()).all())
    by_parent: dict = {}                        # 按 parent_id 分组
    for c in all_visible:
        pid = c.parent_id or None               # None = 顶级评论
        by_parent.setdefault(pid, []).append(c)
    top_comments = by_parent.get(None, [])      # 顶级评论列表
    comments_tree = [                           # 组装成树：每条顶级评论挂自己的回复
        {'comment': c, 'replies': by_parent.get(c.id, [])}
        for c in top_comments]
    comment_count = len(all_visible)            # 评论总数直接取列表长度（内存 O(1)）

    like_count = Like.query.filter_by(post_id=post.id).count()    # 点赞总数（一次 count）

        post_id=post.id, fingerprint=like_fingerprint()
    ).first() is not None

    return render_template('post_detail.html', post=post, html_content=html,

                           comments_tree=comments_tree, comment_count=comment_count,
                           like_count=like_count, liked=liked)
```

**逐行要点：**① `<int:post_id>` 的 `int` 转换器保证 `/post/abc/` 直接 404 而不进函数；② `rendered_html` 懒回填是**缓存策略**：文章保存时渲染好存库（见后台路由），老数据首次访问再补渲染——把「每次访问渲染几十毫秒」变成「只渲染一次」，`synchronize_session=False` 跳过 ORM 会话同步进一步提速；③ prev/next 用 id 大小定位相邻文章，都过滤 `published`；④ **评论 N\+1 问题修复**是本项目一个典型性能优化：旧代码「1 次查顶级 \+ N 次查回复」，评论 50 条就要 51 次数据库查询；改成 1 次查全部 \+ Python 字典分组，永远 1 次查询；⑤ `_vc_count_view` 只做内存 \+1，绝不在请求里 commit——这是「浏览量不卡页面」的根基。

```
            return redirect(url_for('post_detail', post_id=post_id) + '#comments')

    is_admin = False                            # 登录已移除：评论一律按访客身份记录
    comment = Comment(post_id=post_id, parent_id=parent_id,
                      nickname=nickname, email=email or None,
                      website=website or None, content=content,
                      is_admin=is_admin, is_visible=True)
    db.session.add(comment)                     # 加入会话（尚未写库）

```

**逐行要点：**① `methods=['POST']` 很关键——评论是「改数据」操作，用 GET 会被浏览器/爬虫随意触发，内容还会出现在网址里；② `(... or '').strip()` 是防 None \+ 去空格的三件套，否则用户只打空格也能提交；③ `parent_id` 用 `type=int` 转换，用户传非数字自动变 None（即当顶级评论）；④ **父评论校验同时带 ****`post_id`** 是防「串台回复」——只校验 id 存在不校验归属，就能把文章 A 的评论当成文章 B 的回复；⑤ 长度截断校验（50/2000）是防御：数据库字段有长度限制，超长写入会抛 `DataError` 整单失败；⑥ `add + commit` 两步是 SQLAlchemy 固定模式——`add` 只是登记，`commit` 才是真写库，忘记 commit 数据会丢。

扩展知识点：**GET vs POST**。GET 是「读取」（网址可见、可缓存、可收藏）；POST 是「提交」（数据在请求体、不可缓存）。铁律：**只读用 GET，改数据用 POST**——这是 Web 安全第一课。

```
@app.route('/post/&lt;int:post_id&gt;/like/', methods=['POST'])
def post_like(post_id):
    """点赞 / 取消点赞（基于访客指纹去重）"""
    post = Post.query.get_or_404(post_id)
    if not post.published:
        return jsonify({'ok': False, 'error': '文章不可点赞'}), 404

    fp = like_fingerprint()                     # 计算用户指纹：sha256(IP + UA)（见模块四）
    existing = Like.query.filter_by(post_id=post_id, fingerprint=fp).first()
    if existing:                                # 已点过 → 这次是「取消赞」
        db.session.delete(existing)             # 物理删除这条点赞记录
        db.session.commit()
        count = Like.query.filter_by(post_id=post_id).count()   # 重新数一遍
        return jsonify({'ok': True, 'liked': False, 'count': count})
    else:                                       # 没点过 → 新增赞
        like = Like(post_id=post_id, fingerprint=fp)
        db.session.add(like)
        db.session.commit()
        count = Like.query.filter_by(post_id=post_id).count()
        return jsonify({'ok': True, 'liked': True, 'count': count})

```

**逐行要点：**① `like_fingerprint()` 用 IP\+浏览器 UA 生成唯一标识——没有登录系统也能「每人一票」；② 这是**切换（toggle）式点赞**：点过再点就是取消，实现方式是**物理删除记录**（不是软删），因为点赞记录量小、没有跨端同步需求，物理删除最干净；③ 每次操作后重新 `count()` 返回给前端，保证数字永远真实；④ 返回 `jsonify` 让前端 fetch 拿到 `{ok, liked, count}`，实现无刷新点赞。**删掉 ****`if not post.published: 404`**** 的后果：**草稿文章也能被点赞，数据污染。

### 文章删除 post\_delete（第 1635\~1646 行）——修复「删除报错」的关键

```
@app.route('/admin/post/&lt;int:post_id&gt;/delete/', methods=['POST'])
def post_delete(post_id):
    post = Post.query.get_or_404(post_id)
    # 先清理该文章的点赞记录（Like 无 relationship 级联，不显式删除会残留孤儿行）
    Like.query.filter_by(post_id=post.id).delete(synchronize_session=False)
    # 评论由 Comment.post relationship 的 cascade='all, delete-orphan' 级联删除
    db.session.delete(post)
    db.session.commit()
    flash('文章已删除', 'info')
    return redirect(url_for('admin_dashboard'))

```

**为什么顺序这么重要（这正是你之前遇到的 ****`NOT NULL constraint failed: comment.post_id`**** 报错的修复）：**数据库里 `comment.post_id` 是 NOT NULL 外键。SQLAlchemy 的 relationship 如果没配级联，删除 Post 时默认会把关联的 Comment\.post\_id 置为 NULL——直接违反 NOT NULL 约束，抛 `IntegrityError`。修复方案分两层：

- **评论层**：在模型 `Comment.post` 的 relationship 上声明 `cascade='all, delete-orphan'`（见模块二），删文章时让 SQLAlchemy 主动先删评论；

- **点赞层**：`Like` 没有声明级联（刻意为之，点赞记录简单），所以这里**显式先删**——`Like.query.filter_by(...).delete(synchronize_session=False)` 是一条批量 DELETE，`synchronize_session=False` 跳过 ORM 会话同步以提速。

**删掉这行 ****`Like...delete`**** 的后果：**点赞表残留「孤儿行」（指向已删除文章的赞），虽然不报错但会污染统计数据、浪费存储；**如果模型没配 cascade 又直接 ****`db.session.delete(post)`**，则立刻复现你见过的 NOT NULL 报错。

### SECRET\_KEY 持久化（第 31\~86 行）——「为什么每次重启不用重新登录」的答案

```
def _ensure_persistent_secret_key(data_dir: str) -> str:
    """把 SECRET_KEY 写入并固定读 DATA_DIR/.session_key。"""
    key_env = os.environ.get('SECRET_KEY')     # ① 环境变量优先（运维注入）
    if key_env:
        return key_env
    os.makedirs(data_dir, exist_ok=True)       # 确保数据目录存在
    path = os.path.join(data_dir, '.session_key')   # 密钥文件路径
    if os.path.isfile(path):                   # ② 已有密钥文件 → 读回
        try:
            with open(path, 'r', encoding='utf-8') as f:
                content = f.read().strip()
            if len(content) >= 32:             # 长度校验（太短视为损坏）
                return content
        except OSError:
            pass                               # 损坏 → 走下面重建
    import secrets as _secrets, tempfile as _tf
    new_key = _secrets.token_hex(48)           # ③ 生成强随机密钥（96 位十六进制）
    try:
        fd, tmp = _tf.mkstemp(prefix='.session_key.', dir=data_dir, text=True)  # 先写临时文件
        with os.fdopen(fd, 'w', encoding='utf-8') as f:
            f.write(new_key + '\n')
        try:
            os.replace(tmp, path)              # 原子替换：要么旧文件要么新文件，不会半截
            return new_key                     # ⚠️ 必须 return！漏写会返回 None
        except OSError:
            try:                              # 并发首次启动：别人先写了就读回来
                with open(path, 'r', encoding='utf-8') as f:
                    return f.read().strip() or new_key
            except OSError:
                return new_key
        ...
    except OSError:
        return new_key                         # 写盘失败也给当次能用的 key
    return new_key                             # 兜底：绝不能返回 None

```

**为什么这么重要：**Flask 的 session（登录状态）是把数据用 SECRET\_KEY 签名后存在用户浏览器的 cookie 里。如果 SECRET\_KEY 每次启动都随机生成，则重启后旧 cookie 的签名校验失败——用户「每次启动都要重新登录」。把密钥落盘到 `.session_key` 文件，重启后读出同一个密钥，登录态就跨重启保持。**原子写入**（先写临时文件再 `os.replace`）保证断电/崩溃也不会留下半个文件。**删掉这段的后果：**① 没有持久密钥时登录态每次重启失效；② 若函数漏写 `return`（代码注释里专门强调的坑），SECRET\_KEY 为 None，所有 session 功能直接 500。

```
def _load_dotenv(path=None):
    """轻量 .env 加载（不依赖 python-dotenv）"""
    env_path = path or os.path.join(BASE_DIR, '.env')
    if not os.path.isfile(env_path):
        return                              # 没有 .env 文件就跳过
    try:
        from dotenv import load_dotenv as _ext_load   # 装了 python-dotenv 就优先用（功能更全）
        _ext_load(dotenv_path=env_path, override=False, encoding='utf-8')

    except Exception:
        pass                                # 没装 → 用下面手写的极简解析

        with open(env_path, 'r', encoding='utf-8') as f:
            for raw in f:
                line = raw.strip()
                if not line or line.startswith('#'):
                    continue                # 空行/注释跳过
                line = re.sub(r'^export\s+', '', line, flags=re.IGNORECASE)  # 兼容 export KEY=V
                m = re.match(r'([A-Za-z_][A-Za-z0-9_]*)\s*=\s*(.*)', line)   # 拆 KEY=VALUE
                if not m:
                    continue
                key, value = m.group(1), m.group(2)
                value = value.strip()
                if len(value) >= 2 and value[0] == value[-1] and value[0] in ('"', "'"):
                    value = value[1:-1]     # 去掉值两侧的引号
                if key not in os.environ:   # 不覆盖已存在的环境变量（.env 语义）
                    os.environ[key] = value
    except OSError:
        pass
```

```
    created_at = db.Column(db.DateTime, default=datetime.datetime.utcnow)  # 创建时间（自动填）
```

**逐行要点：**① `nullable=False` 是数据库层的「必填约束」——不填直接抛 IntegrityError，比 Python 层校验更硬；② `default=datetime.datetime.utcnow` 注意是**函数引用不是调用**（没有括号）——这样每行插入时执行一次函数取「当前时刻」，如果写 `utcnow()` 则所有行共用启动时刻；③ `onupdate` 是「修改时自动刷新」——任何 UPDATE 都会自动更新 updated\_at，不用手写；④ `lazy='dynamic'` 表示 `tag.posts` 返回的是「查询对象」而非列表，可以继续 `.filter_by(published=True)` 链式查询；⑤ `is_deleted` 是**软删墓碑**——删除不物理删行，只置 1，配合多端同步（墓碑会被同步到其它设备）；⑥ `rendered_html` 缓存列是这个项目的性能核心之一。

扩展知识点：**default vs onupdate**。default 只在 INSERT（插入）时生效；onupdate 在 UPDATE（修改）时自动重新执行。两者都指向「函数」，ORM 会在对应时机调用它取当前时间。

```
    replies = db.relationship('Comment', backref=db.backref('parent', remote_side=[id]),
                              lazy='dynamic', foreign_keys=[parent_id])
    # 自关联：replies 是这条评论的回复；remote_side=[id] 指明"父"的一端是 id 字段

    @property
    def avatar_url(self):
        """Gravatar 头像（基于 email md5），失败回退到 DiceBear"""
        import hashlib
        if self.email:
            h = hashlib.md5(self.email.strip().lower().encode('utf-8')).hexdigest()
            return f'https://cravatar.cn/avatar/{h}?d=identicon&amp;s=80'   # 邮箱 md5 → 全球统一头像
```

**这段是全项目最重要的模型代码之一**，直接对应你之前遇到的删除文章报错：

- **报错根源**：`comment.post_id` 是 `nullable=False` 的 NOT NULL 外键。SQLAlchemy 删除 Post 时，如果 relationship 没配级联，会先执行「把子表外键置 NULL」再删父行 → 违反 NOT NULL 约束 → `IntegrityError: NOT NULL constraint failed: comment.post_id`；

### 备份资产清单：sha256 去重（第 242\~390 行）

```
def _assets_digest(files):                 # 计算「资产指纹」：所有文件的 (路径,大小,sha256) 拼起来再哈希
    """用于判断本次备份与上次内容是否一致，一致则跳过整个备份（省流量）"""
    parts = []
    for f in sorted(files, key=lambda x: x.get('path', '')):   # 排序保证顺序稳定
        parts.append(f"{f.get('path')}|{f.get('size')}|{f.get('sha256','')}")
    return hashlib.sha256(';'.join(parts).encode('utf-8')).hexdigest()

def _sha256_file(path, chunk=1024*1024):   # 大文件分块算哈希（1MB 一块，不占内存）
    h = hashlib.sha256()
```

### 冲突合并：LWW 规则（第 363\~506 行）

```
def _apply_one_record(session, model, rec):
    """合并一条远端记录到本地。
    规则：updated_at 晚的赢（LWW = Last Write Wins，最后写入者胜）
    """
    pk = rec.get('id')                     # 主键定位本地记录
    local = session.query(model).filter_by(id=pk).first() if pk else None
    if local is None:
        # 本地没有 → 直接插入（还要处理墓碑：is_deleted=True 的记录插进来等于"确认删除"）
        if rec.get('is_deleted'):
            return 0                       # 远端说已删除且本地没有 → 无事发生
        session.add(model(**{k: v for k, v in rec.items() if k != '_provenance'}))
        return 1
    # 本地有 → 比时间
    remote_ts = rec.get('updated_at') or ''
    local_ts = _model_updated_at(local) or ''
    if remote_ts &gt; local_ts:               # 远端更新 → 覆盖本地
        for k, v in rec.items():
            if k == '_provenance':
                continue
            setattr(local, k, v)           # 逐字段覆盖（包括 is_deleted 墓碑）
        return 1
    return 0                               # 本地更新 → 保留本地（远端被忽略）

```

**为什么 LWW：**两台电脑同时改同一篇文章时，没有「谁对谁错」的绝对标准，只能约定一个规则——「最后保存的人赢」最简单可靠，也是 Chrome 云同步的合并思路。缺点（会丢一方修改）由「简单可靠」来补偿，个人博客场景完全够用。**墓碑传播：**`is_deleted=True` 的记录会被同步到其它设备，让删除操作也在所有设备生效——这就是模型里都有 `is_deleted` 字段的原因。

```
def _tpl_localtime(dt):
    """naive UTC datetime -> 本地时间（模板显示用，避免 8 小时时差/跨日偏移）"""
    if dt is None:
        return dt
    try:
        if getattr(dt, 'tzinfo', None) is None:
            dt = dt.replace(tzinfo=datetime.timezone.utc)
        return dt.astimezone()
    except Exception:
        return dt

    """Jinja 无内置 strftime 过滤器；本地化后的 datetime 按格式输出"""

        return ''
    try:
        return dt.strftime(fmt)
    except Exception:
        return ''

@app.template_filter('localdt')
def _tpl_localdt(value, fmt='%Y-%m-%d %H:%M:%S'):
    """naive UTC ISO 字符串 -> 本地时间格式化字符串（同步页 finished_at 等）"""
    if not value:
        return ''
    try:
        if isinstance(value, datetime.datetime):
            dt = value
        else:
            dt = datetime.datetime.fromisoformat(str(value))
        if dt.tzinfo is None:
            dt = dt.replace(tzinfo=datetime.timezone.utc)
        return dt.astimezone().strftime(fmt)
    except Exception:
        return str(value)
```

**逐行要点：**① `@app.template_filter('localtime')` 把函数注册成 Jinja 过滤器，模板里写 `{{ post.created_at | localtime }}` 即可，所有页面共用；② 核心动作是 `replace(tzinfo=utc)` + `astimezone()`：先把无时区的 naive UTC 时间「声明」为 UTC，再转成系统本地时区（东八区）——`astimezone()` 不带参数时用系统默认时区，正好适配「在国内运行就是北京时间」的场景；③ 三个过滤器分工：`localtime` 处理 datetime 对象、`strftime` 补 Jinja 缺失的格式化能力、`localdt` 处理同步页 `finished_at` 这类 **ISO 字符串**；④ 全部 try/except 兜底，任一环节异常都返回原值/空串，**绝不因显示时间把整个页面搞挂**；⑤ 同步日志页的 `updated_at` 也统一走这套过滤器，所以「同步时间显示」与「文章时间显示」的时区口径完全一致。

```
def generate_search_snippet(content, query, length=160):
    """从正文中提取包含搜索词的片段（用于搜索结果高亮预览）"""
    if not query or not content:
        return ''
    text = re.sub(r'```.*?```', '', content, flags=re.S)   # 同样先清代码块和标记
    text = re.sub(r'[#*_`&gt;\[\]!()]', '', text)
    text = re.sub(r'\s+', ' ', text).strip()
    idx = text.lower().find(query.lower())                 # ④ 不区分大小写找搜索词位置
    if idx == -1:
        return text[:length] + '...' if len(text) &gt; length else text   # 没找到 → 返回开头
    half = max(0, length // 2 - len(query) // 2)           # ⑤ 以搜索词为中心计算截取窗口
    start = max(0, idx - half)
    end = min(len(text), idx + len(query) + (length - half))
    snippet = text[start:end]                              # ⑥ 截出「搜索词在中间」的片段
    if start &gt; 0:
        snippet = '...' + snippet                          # 前后加省略号提示被截断
    if end &lt; len(text):
        snippet = snippet + '...'
    return snippet

```

**为什么「先删代码块再删符号」：**如果直接删 `#`，会把代码注释里的 `#` 也删掉；先整块删掉代码块再处理剩下的 Markdown 标记，摘要就不会被代码污染。**搜索片段「以词为中心」的妙处：**用户搜「SQLite」，返回的片段从 SQLite 出现的位置截取，而不是从文章开头——用户一眼看到命中内容，体验远好于只显示开头。

```
def parse_front_matter(text):
    """解析 Markdown / YAML 文件中的 front-matter（--- 包裹的 YAML 头部）。
    返回 (metadata_dict, body_content)"""
    if text.startswith('---'):             # 只有以 --- 开头才可能是 front-matter
        match = re.match(r'^---\s*\n(.*?)\n---\s*\n?(.*)', text, flags=re.S)
        # 非贪婪 (.*?)：匹配第一个 --- 到第二个 --- 之间的内容
        if match:
            try:
                meta = yaml.safe_load(match.group(1)) or {}   # YAML 头 → 字典
                body = match.group(2)      # 第二个 --- 之后是正文
                return meta, body
            except yaml.YAMLError:
                pass                       # YAML 解析失败 → 当作普通 Markdown 处理
    return {}, text                        # 没有 front-matter → 原样返回，元数据为空

```

**要点：**① front\-matter 是 Markdown 文件开头的 `---` 包裹区，用于携带标题、标签、日期等元信息——批量导入文章（模块一 upload\_file）就靠它；② **解析失败静默回退**——用户手抖写错 YAML 时，整篇文件仍能作为普通 Markdown 导入，只是没有元信息，不崩；③ `.*?` 非贪婪匹配是这段的核心：贪婪匹配 `.*` 会吞掉第一个 `---` 直接匹配到最后一个，把正文全吃掉。

### 窗口居中 \_compute\_centered\_position（第 105\~141 行）——v2.1 新增

```
def _compute_centered_position(width: int, height: int):
    """计算让窗口在屏幕正中央的 (x, y)。

    Windows 上按 DPI 感知返回逻辑像素（DIP）：
      · GetSystemMetrics 返回物理像素；
      · pywebview / WebView2 的 x/y 是 DIP；
      · 若系统缩放（如 125%/150%）非 100%，必须先把物理像素换算成 DIP，
        否则窗口会偏到右下，达不到"每次打开都在屏幕中间"的效果。
    非 Windows / 计算失败时返回 None，调用方保持默认定位。
    """
    try:
        if sys.platform != 'win32':
            return None
        import ctypes
        user32 = ctypes.windll.user32
        SM_CXSCREEN = 0
        SM_CYSCREEN = 1
        sw = user32.GetSystemMetrics(SM_CXSCREEN)   # 屏幕宽度（物理像素）
        sh = user32.GetSystemMetrics(SM_CYSCREEN)   # 屏幕高度（物理像素）
        if sw <= 0 or sh <= 0:
            return None
        # DPI 换算：逻辑像素 = 物理像素 ÷ (系统缩放比例)
        dpi = 96
        try:
            dpi = int(user32.GetDpiForSystem())
        except Exception:
            dpi = 96
        if dpi <= 0:
            dpi = 96
        scale = dpi / 96.0
        sw = int(sw / scale)               # 物理像素 → DIP
        sh = int(sh / scale)
        x = max(0, (sw - width) // 2)      # 水平居中：屏幕宽减窗口宽的一半
        y = max(0, (sh - height) // 2)     # 垂直居中：屏幕高减窗口高的一半
        return x, y
    except Exception:
        return None
```

**逐行要点：**① 窗口的 `x/y` 坐标由 `_build_create_window_kwargs`（第 326\~375 行）调用本函数算好后传进 `webview.create_window`——所以**每次打开主窗口都在屏幕正中央，不随上次关闭位置漂移**（这就是「每次打开位置都不一样」的修复）；② **DPI 换算**是这里的灵魂：Windows 的 `GetSystemMetrics` 返回物理像素，而 pywebview/WebView2 的坐标是逻辑像素（DIP）。系统缩放 125% 时 `scale=1.25`，不除以它窗口会偏到右下——这是「为什么我算了居中还是偏」的常见答案；③ 返回 `None` 时调用方保持默认定位（非 Windows 或计算失败不硬来）。

### 窗口大小记忆：拖拽调整后自动记住（第 144\~324 行）——v2.8.3 新增

> 背景：v2.1 只有「每次启动自动居中」，窗口大小固定 1280×820；用户希望「拖拽调整大小后，下次启动沿用最新尺寸」。v2.8.3 新增一整块「窗口大小记忆」模块，核心是**监听 resize 事件 → 防抖写盘 → 下次启动读盘恢复**。

```
# ---------- 窗口大小记忆（拖拽调整后自动记住，下次启动沿用）----------
_SIZE_MIN_W, _SIZE_MIN_H = 960, 640          # 与 create_window 的 min_size 一致
_SIZE_DEFAULT = (1280, 820)                  # 首次启动/无记忆时默认
_SIZE_SAVE_DEBOUNCE = 0.8                    # 拖拽结束 0.8s 无变化才写盘
_size_timer = None                            # debounce 定时器
_size_timer_lock = threading.Lock()
_size_flush_callback = None                   # 关闭窗口前立即刷新的回调

def _window_state_path() -> str:
    from config import Config as _Cfg
    return os.path.join(_Cfg.DATA_DIR, 'window_state.json')

def _load_window_size():
    """读取上次保存的窗口大小（DIP）。缺失/非法/超屏一律回退默认。"""
    try:
        p = _window_state_path()
        if not os.path.isfile(p):
            return None
        with open(p, 'r', encoding='utf-8') as f:
            d = json.load(f)
        w = int(d.get('width', 0)); h = int(d.get('height', 0))
        if w < _SIZE_MIN_W or h < _SIZE_MIN_H:
            return None
        scr = _screen_dip_size()
        if scr and (w > scr[0] or h > scr[1]):
            return None
        return w, h
    except Exception:
        return None

def _save_window_size(width: int, height: int) -> None:
    """原子写入窗口大小（失败静默，不影响主流程）。"""
    try:
        p = _window_state_path()
        d = os.path.dirname(p)
        if d and not os.path.isdir(d):
            os.makedirs(d, exist_ok=True)
        tmp = p + '.tmp'
        with open(tmp, 'w', encoding='utf-8') as f:
            json.dump({'width': int(width), 'height': int(height)}, f, ensure_ascii=False)
        os.replace(tmp, p)
    except Exception:
        pass

def _bind_window_size_save(window) -> None:
    """监听窗口 resized 事件：拖拽结束后（防抖）把最新大小写入记忆文件。
    多次调整始终以最后一次为准；最大化/最小化不覆盖记录。"""
    global _size_timer, _size_flush_callback
    def _flush():
        global _size_timer
        with _size_timer_lock:
            if _size_timer is not None:
                _size_timer.cancel(); _size_timer = None
        rect = _restored_window_size_dip(window)
        if rect:
            w, h = rect
        else:
            w = getattr(window, 'width', None); h = getattr(window, 'height', None)
            if not w or not h:
                return
        if w < _SIZE_MIN_W or h < _SIZE_MIN_H:
            return
        _save_window_size(w, h)
    def _on_resized(*_args):
        global _size_timer
        with _size_timer_lock:
            if _size_timer is not None:
                _size_timer.cancel()
            _size_timer = threading.Timer(_SIZE_SAVE_DEBOUNCE, _flush)
            _size_timer.daemon = True
            _size_timer.start()
    try:
        window.events.resized += _on_resized
    except Exception:
        pass
    _size_flush_callback = _flush

def _flush_window_size_now() -> None:
    """窗口关闭前调用：立即把当前大小写入记忆（不被防抖吞掉）。"""
    if _size_flush_callback is not None:
        try:
            _size_flush_callback()
        except Exception:
            pass
```

**逐行要点：**① **防抖（debounce）**是这块设计的核心：拖拽过程中 `resized` 事件会高频触发（每帧都可能），如果每次都写盘，磁盘 IO 太频繁。方案是 `threading.Timer(0.8, _flush)`——每次 resize 都重置 0.8 秒计时器，**停止拖拽 0.8 秒后才真正写盘**；② `_load_window_size` 有三道防线：文件不存在回退默认、小于最小尺寸（960×640）回退默认、**超过屏幕尺寸回退默认**（防止换了小屏幕后窗口超出可视区域）；③ `_save_window_size` 用「写临时文件 + `os.replace`」实现原子替换——避免写到一半程序崩溃留下损坏的 JSON；④ `window.events.resized` 是 pywebview 提供的事件钩子，`+= _on_resized` 订阅；`_restored_window_size_dip` 用 Win32 `GetWindowPlacement` 取「还原状态」尺寸——**最大化/最小化时不把全屏尺寸误存**；⑤ `_flush_window_size_now` 在关闭窗口前调用：如果用户调整完大小**不到 0.8 秒就关窗**，防抖定时器还没来得及触发，关闭前强制 flush 一次，保证最后一次调整不丢；⑥ `_screen_dip_size` / `_restored_window_size_dip` 都做 DPI 换算（物理像素 ÷ 缩放比例），与居中函数的 DIP 口径一致，避免 125% 缩放下「存进去的尺寸和显示尺寸对不上」。

```
    s = socket.socket(...)
    try:

        pass
    finally:
        s.close()
    for p in range(max(preferred + 1, start), end):   # ③ 被占才向后扫描
        ...bind 探测...
```

**托盘原理（大白话）：**Windows 的托盘图标本质上是一个「隐形消息窗口 \+ 一个系统图标」。程序创建了一个隐藏窗口（`_tray_thread_main` 里 `CreateWindowExW`），把图标交给 Shell（`Shell_NotifyIconW`），并告诉系统「鼠标点图标时，往我这个窗口发 `WM_TRAYICON` 消息」。之后程序进入**消息循环**（`GetMessageW`），收到消息就调用 `_tray_wndproc` 分发：左键 → `_tray_show_activate()`（显示并置前）；右键 → `_tray_show_menu()`（弹菜单）。

```
def _tray_wndproc(hwnd, msg, wparam, lparam):
    """托盘消息窗口过程：处理图标回调与菜单命令"""
    if msg == WM_TRAYICON:
        ev = lparam &amp; 0xFFFF      # 低 16 位 = 鼠标事件码（高 16 位是图标 ID）
        # v4 通知协议：右键发 WM_CONTEXTMENU；v3 兼容：右键发 WM_RBUTTONUP —— 都处理
        if ev in (WM_RBUTTONUP, WM_CONTEXTMENU):
            _tray_show_menu()      # 右键 → 菜单
        elif ev in (WM_LBUTTONUP, WM_LBUTTONDBLCLK):
            _tray_show_activate()  # 左键单击/双击 → 打开应用
    elif msg == WM_COMMAND:
        _tray_handle_command(wparam &amp; 0xFFFF)   # 菜单命令分发
    return DefWindowProcW(hwnd, msg, wparam, lparam)   # 其余消息交默认处理

        return    # 右键会同时触发 WM_RBUTTONUP 和 WM_CONTEXTMENU 两条消息，300ms 去重

    user32.AppendMenuW(hmenu, MF_STRING, IDM_TRAY_QUIT, '退出博客')
```
